cmake_minimum_required(VERSION 3.16)
project(cilogg_plugin_example LANGUAGES CXX)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core)
add_executable(cilogg_plugin_example main.cpp)
target_compile_features(cilogg_plugin_example PRIVATE cxx_std_17)
target_include_directories(cilogg_plugin_example PRIVATE ${CILOGG_PLUGIN_SDK_DIR})
target_link_libraries(cilogg_plugin_example PRIVATE Qt${QT_VERSION_MAJOR}::Core)
configure_file(plugin.json plugin.json COPYONLY)
